const canvas = document.getElementById("gameCanvas");
const ctx = canvas.getContext("2d");

// Game State Variables
let score = 0;
let lives = 3;
let isGameRunning = false;
let animationFrameId;

// Player Properties
const player = {
    x: canvas.width / 2 - 15,
    y: canvas.height - 50,
    width: 30,
    height: 30,
    speed: 5,
    dx: 0
};

// Controls State
const keys = {
    left: false,
    right: false,
    space: false
};

// Projectiles & Enemies
let bullets = [];
let asteroids = [];
let lastShotTime = 0;

// Key Event Listeners
document.addEventListener("keydown", (e) => {
    if (e.key === "ArrowLeft" || e.key === "a") keys.left = true;
    if (e.key === "ArrowRight" || e.key === "d") keys.right = true;
    if (e.key === " " || e.code === "Space") {
        keys.space = true;
        e.preventDefault();
    }
});

document.addEventListener("keyup", (e) => {
    if (e.key === "ArrowLeft" || e.key === "a") keys.left = false;
    if (e.key === "ArrowRight" || e.key === "d") keys.right = false;
    if (e.key === " " || e.code === "Space") keys.space = false;
});

// Start / Restart Game
function startGame() {
    score = 0;
    lives = 3;
    bullets = [];
    asteroids = [];
    player.x = canvas.width / 2 - 15;
    
    document.getElementById("score").innerText = score;
    document.getElementById("lives").innerText = lives;
    document.getElementById("overlay").style.display = "none";
    
    isGameRunning = true;
    gameLoop();
}

// Game Loop
function gameLoop() {
    if (!isGameRunning) return;

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    updatePlayer();
    updateBullets();
    updateAsteroids();
    checkCollisions();

    drawPlayer();
    drawBullets();
    drawAsteroids();

    animationFrameId = requestAnimationFrame(gameLoop);
}

// Player Movement & Shooting
function updatePlayer() {
    if (keys.left && player.x > 0) player.x -= player.speed;
    if (keys.right && player.x + player.width < canvas.width) player.x += player.speed;

    // Fire laser with cooldown
    const now = Date.now();
    if (keys.space && now - lastShotTime > 200) {
        bullets.push({
            x: player.x + player.width / 2 - 2,
            y: player.y,
            width: 4,
            height: 10,
            speed: 7
        });
        lastShotTime = now;
    }
}

function drawPlayer() {
    ctx.fillStyle = "#00ffcc";
    // Draw triangular spaceship
    ctx.beginPath();
    ctx.moveTo(player.x + player.width / 2, player.y);
    ctx.lineTo(player.x, player.y + player.height);
    ctx.lineTo(player.x + player.width, player.y + player.height);
    ctx.closePath();
    ctx.fill();
}

// Bullets Handler
function updateBullets() {
    for (let i = bullets.length - 1; i >= 0; i--) {
        bullets[i].y -= bullets[i].speed;
        if (bullets[i].y < 0) bullets.splice(i, 1);
    }
}

function drawBullets() {
    ctx.fillStyle = "#ff0055";
    bullets.forEach(b => ctx.fillRect(b.x, b.y, b.width, b.height));
}

// Asteroids Handler
function updateAsteroids() {
    // Randomly spawn asteroid
    if (Math.random() < 0.02) {
        asteroids.push({
            x: Math.random() * (canvas.width - 30),
            y: -30,
            size: 25 + Math.random() * 10,
            speed: 2 + Math.random() * 2
        });
    }

    for (let i = asteroids.length - 1; i >= 0; i--) {
        asteroids[i].y += asteroids[i].speed;

        // Passed bottom edge
        if (asteroids[i].y > canvas.height) {
            asteroids.splice(i, 1);
            takeDamage();
        }
    }
}

function drawAsteroids() {
    ctx.fillStyle = "#ffcc00";
    asteroids.forEach(a => {
        ctx.beginPath();
        ctx.arc(a.x + a.size / 2, a.y + a.size / 2, a.size / 2, 0, Math.PI * 2);
        ctx.fill();
    });
}

// Collision Logic
function checkCollisions() {
    for (let i = asteroids.length - 1; i >= 0; i--) {
        const a = asteroids[i];

        // Laser hits asteroid
        for (let j = bullets.length - 1; j >= 0; j--) {
            const b = bullets[j];
            if (
                b.x < a.x + a.size &&
                b.x + b.width > a.x &&
                b.y < a.y + a.size &&
                b.y + b.height > a.y
            ) {
                asteroids.splice(i, 1);
                bullets.splice(j, 1);
                score += 10;
                document.getElementById("score").innerText = score;
                break;
            }
        }

        // Asteroid hits player ship
        if (
            a &&
            player.x < a.x + a.size &&
            player.x + player.width > a.x &&
            player.y < a.y + a.size &&
            player.y + player.height > a.y
        ) {
            asteroids.splice(i, 1);
            takeDamage();
        }
    }
}

function takeDamage() {
    lives--;
    document.getElementById("lives").innerText = lives;
    if (lives <= 0) {
        gameOver();
    }
}

function gameOver() {
    isGameRunning = false;
    cancelAnimationFrame(animationFrameId);
    document.getElementById("overlayTitle").innerText = "GAME OVER";
    document.getElementById("overlaySubtitle").innerText = `Final Score: ${score}`;
    document.getElementById("startBtn").innerText = "PLAY AGAIN";
    document.getElementById("overlay").style.display = "flex";
}
